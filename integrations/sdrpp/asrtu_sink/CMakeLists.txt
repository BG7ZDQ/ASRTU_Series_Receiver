cmake_minimum_required(VERSION 3.16)
project(asrtu_sink LANGUAGES CXX)
set(SDRPP_CORE_ROOT "" CACHE PATH "SDR++ core directory (contains src/module.h)")
set(SDRPP_CORE_LIBRARY "" CACHE FILEPATH "Matching sdrpp_core import library for standalone builds")
set(SDRPP_DEPENDENCY_INCLUDE_DIR "" CACHE PATH "SDR++ dependency headers (volk, fftw3)")
set(SDRPP_VOLK_LIBRARY "" CACHE FILEPATH "Matching volk import library for standalone builds")
if(NOT EXISTS "${SDRPP_CORE_ROOT}/src/module.h")
    message(FATAL_ERROR "Set SDRPP_CORE_ROOT to the matching SDR++ source tree's core directory")
endif()
add_library(asrtu_sink SHARED src/main.cpp src/shared_bridge.h)
target_include_directories(asrtu_sink PRIVATE
    "${SDRPP_CORE_ROOT}/src" "${SDRPP_CORE_ROOT}/src/imgui")
if(TARGET sdrpp_core)
    target_link_libraries(asrtu_sink PRIVATE sdrpp_core)
elseif(WIN32)
    if(NOT EXISTS "${SDRPP_CORE_LIBRARY}" OR NOT EXISTS "${SDRPP_VOLK_LIBRARY}")
        message(FATAL_ERROR "Standalone builds need SDRPP_CORE_LIBRARY and SDRPP_VOLK_LIBRARY from the target SDR++ build")
    endif()
    target_include_directories(asrtu_sink PRIVATE "${SDRPP_DEPENDENCY_INCLUDE_DIR}")
    target_link_libraries(asrtu_sink PRIVATE "${SDRPP_CORE_LIBRARY}" "${SDRPP_VOLK_LIBRARY}")
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(VOLK REQUIRED IMPORTED_TARGET volk)
    pkg_check_modules(FFTW REQUIRED fftw3f)
    target_include_directories(asrtu_sink PRIVATE ${FFTW_INCLUDE_DIRS})
    target_link_libraries(asrtu_sink PRIVATE PkgConfig::VOLK)
    if(APPLE)
        target_link_options(asrtu_sink PRIVATE "-undefined" "dynamic_lookup")
    endif()
endif()
target_compile_features(asrtu_sink PRIVATE cxx_std_17)
target_compile_definitions(asrtu_sink PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _USE_MATH_DEFINES)
if(MSVC)
    target_compile_options(asrtu_sink PRIVATE /EHsc /utf-8)
endif()
set_target_properties(asrtu_sink PROPERTIES PREFIX "")
install(TARGETS asrtu_sink RUNTIME DESTINATION modules LIBRARY DESTINATION modules)
